# --- plan step auditing ---
# With plan_audit=1, every mutating execute_plan step is verified against the
# real environment by the readonly `auditor` sub-agent before the plan moves on,
# and a failed verification halts the plan. Off by default: it costs one extra
# model call per mutating step, drawn from the same turn budget as the work
# itself. Worth enabling for unattended runs (gateway, cron) where no operator
# is watching, which is exactly where a silently half-done plan does damage.
# plan_audit=0
# With auditing on, a write that FAILS verification is restored to its exact
# pre-step bytes: only verified state persists. Nothing outside the failed step
# is touched, and tools with no undo (shell, docker, cron) say so instead of
# implying a rollback that did not happen.

# --- persistent memory ---
# On by default. With memory=1 the agent does two extra things per turn:
#
#   before the turn  the last thing YOU typed is searched against stored
#                    memories, and the top few are added to that turn's system
#                    prompt as context
#   after the answer one extra model call distils durable facts from the
#                    exchange and stores them, deduped against what is known
#
# Retrieval runs two independent searches and fuses them with Reciprocal Rank
# Fusion: SQLite FTS5/BM25 for words (exact flags, filenames, error strings) and
# cosine similarity over embeddings for meaning (finds "uv" from "package
# manager"). Their scores are not comparable, so RRF fuses the ranks instead --
# which is why the vector leg only has to be roughly right, and why a 274MB
# embedder is enough.
#
# The installers pull the embedding model, so semantic recall works from the
# first turn. `/memory` shows what it holds and what the last learning call cost;
# `/memory off` stops both habits without deleting anything, and
# memory_capture=0 below keeps recall while stopping new writes.
#
# Only what YOU type can become a memory or trigger a recall. Tool output -- web
# pages, shell results, file contents -- never does, and a recalled memory can
# never authorise a tool call: check_permission does not read them.
memory=1

# --- web search ---
# One web_search tool, four interchangeable backends, in priority order:
#   searxng  default   self-hosted, no key   (`/search setup` provisions it)
#   ddgs     fallback  keyless, bundled — always available, needs no setup
#   tavily   optional  needs TAVILY_API_KEY — takes first priority once set
#   exa      optional  needs EXA_API_KEY — takes priority, behind tavily
# Adding an API key is the signal to prefer that backend, so a configured
# tavily/exa outranks both keyless ones.
#
# web_search_provider decides which one serves:
#
#   auto       (default) At startup, probe and pick the highest-priority
#              backend that can actually serve: a keyed tavily/exa first, else
#              SearXNG if it ANSWERS, else ddgs. The winner is then pinned for
#              the session — auto decides *which* backend, it does not fall
#              through mid-session.
#   <name>     Pin exactly one of searxng/ddgs/tavily/exa. No auto-selection,
#              no fallback. `/search use <name>` writes this for you.
#   (unset)    Use the whole chain and fall through at call time: a backend
#              that fails (instance down, rate limited) hands off to the next,
#              so web search keeps working — at the cost of a prompt per
#              search, since it may reach a public provider.
#
# Why auto pins instead of staying dynamic: web_search_no_prompt=1 below only
# takes effect while a LOCAL SearXNG is pinned (see
# _local_searxng_no_prompt_enabled), because approval-free search is safe only
# when the query cannot leave your network. So with auto: SearXNG up means
# silent searches, and SearXNG down means ddgs serves instead and each search
# asks — because those queries do go to a third party.
#
# Run `/search status` to see the live chain and `/search doctor` to diagnose.
web_search_provider=auto
#
# SearXNG instance. A remote instance must use https:// — plaintext http:// is
# accepted only for loopback/private hosts. The host must appear in
# ssrf_allow_hosts above, or requests to it are blocked as internal.
search_base_url=http://192.168.3.67:8888/search?q=
# Skip approval only for explicitly pinned loopback or explicitly allowlisted
# private-LAN SearXNG. It never permits a fallback to a public provider.
# Credentials and direct personal identifiers in search queries remain blocked.
web_search_no_prompt=1
#
# Results per search (default 5, max 20):
# web_search_results=5

# Optional backends. Add a key and they join the chain automatically; leave them
# unset and they stay out of it entirely. Keys live in the .env store next to
# this file, NEVER here:
#   TAVILY_API_KEY   agent-optimized results with citations  (https://tavily.com)

# The keyless ddgs fallback needs nothing: it ships with agent8088 and is used
