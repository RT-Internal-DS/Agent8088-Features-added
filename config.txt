# Collapse SearXNG's verbose JSON (engines, positions, score, parsed_url…) down to
# title/url/snippet. Keeps search results from eating the context window — which
# matters a lot for smaller local models.
tool_filter.web_search=if (.results | length) == 0 then "No results." else (.results[:6][] | "• \(.title)\n  \(.url)\n  \(.content // "")") end

# --- search fallbacks (optional; used when the LAN SearXNG is unreachable) ---
# Tavily: agent-optimized, returns clean ranked content. Free tier ~1000/mo.
# Get a key at https://tavily.com  then uncomment tavily_api_key.
# tavily_api_key=tvly-...
tool_headers.web_search_tavily=Content-Type: application/json;;Authorization: Bearer {tavily_api_key}
tool_body.web_search_tavily={"query": "{query}", "max_results": 6, "search_depth": "basic", "include_answer": true}
tool_filter.web_search_tavily=(if .answer then "Answer: \(.answer)\n" else "" end) + ([.results[]? | "• \(.title)\n  \(.url)\n  \(.content // "")"] | join("\n"))

# Exa: semantic/neural search. Free tier ~20,000/mo (largest of the hosted APIs).
# Get a key at https://exa.ai  then uncomment exa_api_key.
# exa_api_key=...
tool_headers.web_search_exa=Content-Type: application/json;;x-api-key: {exa_api_key}
tool_body.web_search_exa={"query": "{query}", "numResults": 6, "contents": {"text": {"maxCharacters": 600}}}
tool_filter.web_search_exa=[.results[]? | "• \(.title)\n  \(.url)\n  \((.text // "") | gsub("\\s+"; " "))"] | join("\n")
